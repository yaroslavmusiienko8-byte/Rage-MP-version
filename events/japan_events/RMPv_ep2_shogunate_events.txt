namespace = shogunate

REPLACE_OR_CREATE:shogunate.2 = { # Debate about Iesada vs. Yoshinobu - Ieyoshi
	type = country_event
	placement = root

	duration = 3

	title = shogunate.2.t
	desc = shogunate.2.d
	flavor = shogunate.2.f

    event_image = {
		video = "ep2_important_japanese_men_negotiating"
	}

	dlc = dlc018

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/faction_icons/faction_japan_tokugawa.dds"

	cooldown = { days = never_fire_again_modifier_time }

	trigger = {
		has_dlc_feature = ep2_content
		game_date >= 1848.1.1
		ruler = {
			has_template = JAP_ieyoshi_tokugawa
		}
	}

	cancellation_trigger = {

	}

	immediate = {
		ruler = {
			save_scope_as = shogun_scope
		}
		ig:ig_armed_forces ?= {
			leader = {
				save_scope_as = armed_forces_leader
			}
		}
		if = {
			limit = {
				NOT = {
					any_scope_character = {
						has_template = JAP_iesada_tokugawa
					}
				}
			}
			create_character = {
				template = JAP_iesada_tokugawa
			}
		}
		if = {
			limit = {
				NOT = {
					any_scope_character = {
						has_template = JAP_yoshinobu_tokugawa
					}
				}
			}
			create_character = {
				template = JAP_yoshinobu_tokugawa
			}
		}
		random_scope_character = {
			limit = {
				has_template = JAP_iesada_tokugawa
			}
			save_scope_as = iesada_scope
		}
		random_scope_character = {
			limit = {
				has_template = JAP_yoshinobu_tokugawa
			}
			save_scope_as = yoshinobu_scope
		}
	}

	option = {
		name = shogunate.2.a
		default_option = yes
		if = {
			limit = {
				scope:iesada_scope ?= {
					is_heir_of_own_country = yes
				}
			}
			custom_tooltip = iesada_stays_heir_tt
		}
		else = {
			set_heir = scope:iesada_scope
		}
		ig:ig_landowners ?= {
			add_modifier = {
				name = modifier_japan_selected_shogun_candidate
				days = short_modifier_time
			}
			leader ?= {
				add_modifier = {
					name = modifier_power_behind_the_shogun
					days = normal_modifier_time
				}
			}
		}
		ig:ig_armed_forces ?= {
			add_modifier = {
				name = modifier_japan_capability_disregarded
				days = normal_modifier_time
			}
		}
	}

	option = {
		name = shogunate.2.b
		trigger = {
			ig:ig_armed_forces ?= {
				is_in_government = yes
			}
			legitimacy >= 50
		}
		show_as_unavailable = {
			always = yes
		}
		highlighted_option = yes
		set_heir = scope:yoshinobu_scope
		if = {
			limit = {
				scope:yoshinobu_scope ?= {
					is_adult = no
				}
			}
			custom_tooltip = yoshinobu_is_the_guy_info_tt
		}
		ig:ig_armed_forces ?= {
			add_modifier = {
				name = modifier_japan_selected_shogun_candidate
				days = short_modifier_time
			}
			leader ?= {
				add_modifier = {
					name = modifier_power_behind_the_shogun
					days = normal_modifier_time
				}
			}
		}
		ig:ig_landowners ?= {
			add_modifier = {
				name = modifier_japan_lineage_disregarded
				days = normal_modifier_time
			}
		}
	}
}
